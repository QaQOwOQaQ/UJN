#include<cmath>
#include<cstdio>
#include<cstring>
#include<iostream>
#include<algorithm>

using namespace std;

const int N = 5e6 + 10;

struct zxc{
	int l, r, v;
}t[N * 20];
int top;
int n, k;
int a[N];
int node[N];

int copy(int node)
{
	top ++;
	t[top] = t[node];
	return top;
} 

int build(int root, int l, int r)
{
	root = ++ top;
	if(l == r)
	{
		t[root].v = a[l];
		return top;
	}
	
	int mid = l + r >> 1;
	
	t[root].l = build(t[root].l, l, mid);
	t[root].r = build(t[root].r, mid + 1, r);
	
	return root;
}

int update(int root, int l, int r, int x, int v)
{
	root = copy(root);
	if(l == r)
	{
		t[root].v = v;
	}
	else 
	{
		int mid = l + r >> 1;
		if(x <= mid) 
			t[root].l = update(t[root].l, l, mid, x, v);
		else 
			t[root].r = update(t[root].r, mid + 1, r, x, v);
	}
	
	return root;
}

int query(int root, int l, int r, int x)
{
	if(l == r)
	{
		return t[root].v;
	}
	
	int mid = l + r >> 1;
	
	if(x <= mid)
		return query(t[root].l, l, mid, x);
	else 
		return query(t[root].r, mid + 1, r, x);
}

int main()
{
	scanf("%d%d", &n, &k);
	for(int i = 1; i <= n; i ++) 
		scanf("%d", &a[i]);
	node[0] = build(0, 1, n);
	
	for(int i = 1; i <= k; i ++)
	{
		int v, op, loc, val;
		scanf("%d%d%d", &v, &op, &loc);
		if(op == 1)
		{
			scanf("%d", &val);
			node[i] = update(node[v], 1, n, loc, val);
		}
		else
		{
			int res = query(node[v], 1, n, loc);
			printf("%d\n", res);
			node[i] = node[v];
		}
	}
	return 0;
}
